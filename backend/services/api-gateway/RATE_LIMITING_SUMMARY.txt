================================================================================
USAGE EXAMPLES
================================================================================

Custom Rate Limit:
  @RateLimit({ window: '5m', max: 10 })
  @Post('action')
  async action() { }

Strict Limit:
  @StrictRateLimit()
  @Post('sensitive')
  async sensitive() { }

Skip Rate Limiting:
  @SkipRateLimit()
  @Get('webhook')
  async webhook() { }
